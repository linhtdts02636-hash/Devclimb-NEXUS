def parse_arguments(raw):
    trimmed = raw.strip()
    if not trimmed:
        return []
    lines = [line for line in trimmed.splitlines() if line.strip()]
    if len(lines) > 1:
        return [json.loads(line) for line in lines]
    return [json.loads(trimmed)]
