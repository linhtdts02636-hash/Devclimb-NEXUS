function parseArguments(raw) {
    const trimmed = raw.trim();
    if (!trimmed) {
        return [];
    }
    const lines = trimmed.split(/\r?\n/).filter((line) => line.trim().length > 0);
    if (lines.length > 1) {
        return lines.map((line) => JSON.parse(line));
    }
    return [JSON.parse(trimmed)];
}
